rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // الأدمن: الإيميل الأساسي، أو إيميل مسجل في مجموعة admins (يضيفهم الأدمن من اللوحة)
    // لازم الإيميل يكون متحقق منه (جوجل بيحقق تلقائياً)، عشان محدش يعمل حساب
    // بإيميل وباسورد بإيميل أدمن من غير تحقق وياخد صلاحياته
    function isAdmin() {
      return request.auth != null && request.auth.token.email_verified == true
        && (request.auth.token.email == "osamaasas0@gmail.com"
            || exists(/databases/$(database)/documents/admins/$(request.auth.token.email)));
    }
    function isTeacher() {
      return request.auth != null
        && exists(/databases/$(database)/documents/teachers/$(request.auth.uid))
        && get(/databases/$(database)/documents/teachers/$(request.auth.uid)).data.active == true;
    }

    // المشرفين: مفتاح المستند هو الإيميل. القراءة والإضافة والحذف للأدمن بس،
    // ومستند الأدمن الأساسي ميتحذفش أبداً، والتعديل مقفول (حذف + إضافة من جديد بدله)
    match /admins/{email} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update: if false;
      allow delete: if isAdmin() && email != "osamaasas0@gmail.com";
    }

    // ملفات المدرسين الكاملة (فيها إيميل وتليفون): الأدمن أو المدرس نفسه بس
    match /teachers/{teacherId} {
      allow get: if isAdmin() || request.auth.uid == teacherId;
      allow list: if isAdmin();
      allow create, delete: if isAdmin();
      allow update: if isAdmin()
        || (isTeacher() && request.auth.uid == teacherId
            && request.resource.data.active == resource.data.active);
    }

    // أسماء ومواد المدرسين للعرض العام في صفحة التسجيل — من غير إيميلات ولا تليفونات
    match /teachersPublic/{teacherId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // فهرس تسجيل عام آمن: مفتاحه الإيميل، وفيه هاش الرقم السري + الحالة + رقم ملف الطالب
    // القراءة بمستند واحد بإيميل معروف فقط، وممنوع سرد الفهرس كله
    match /registrationLookup/{lookupId} {
      allow get: if true;
      allow list: if false;
      allow create: if isAdmin() || isTeacher()
        || (request.resource.data.status == "pending"
            && request.resource.data.teacherIds is list);
      allow update, delete: if isAdmin() || isTeacher();
    }

    // المواد الدراسية: القراءة للأدمن والمدرسين، والكتابة للأدمن بس
    match /subjects/{subjectId} {
      allow read: if isAdmin() || isTeacher();
      allow write: if isAdmin();
    }

    // الطلاب: ممنوع أي قراءة عامة
    match /students/{studentId} {
      // مستند واحد: الأدمن، أو مدرس من مدرسين الطالب، أو الطالب نفسه.
      // المطابقة بالإيميل شغالة بس قبل ما الطالب يربط ملفه بحسابه (authLinked):
      // كده حساب بإيميل وباسورد غير متحقق بنفس إيميل طالب ميقدرش يقرا ملفه
      // بعد الربط، والربط نفسه محمي بالرقم السري في registrationLookup
      allow get: if isAdmin()
        || (isTeacher() && request.auth.uid in resource.data.teacherIds)
        || (request.auth != null && request.auth.uid == studentId)
        || (request.auth != null && request.auth.token.email == resource.data.email
            && resource.data.authLinked != true);
      // السرد: الأدمن يشوف الكل، والمدرس (استعلاماته في الكود مقيدة بطلابه بس)
      allow list: if isAdmin() || isTeacher();
      // إنشاء: تسجيل جديد بحالة pending، أو الطالب بينقل بياناته لحسابه عند أول دخول.
      // في النقل: لازم الفهرس العام يكون حالته approved (مدرس أو أدمن بس اللي يعمل كده)
      // وبيانات المستند الجديد تطابق اللي في الفهرس، فمحدش يقدر يوافق على نفسه
      // أو يضيف مدرسين من عنده.
      allow create: if isAdmin() || isTeacher()
        || request.resource.data.status == "pending"
        || (request.auth != null && request.auth.uid == studentId
            && request.resource.data.email == request.auth.token.email
            && get(/databases/$(database)/documents/registrationLookup/$(request.auth.token.email)).data.status == "approved"
            && request.resource.data.status == "approved"
            && request.resource.data.teacherIds == get(/databases/$(database)/documents/registrationLookup/$(request.auth.token.email)).data.teacherIds);
      // تعديل: الأدمن، أو مدرس من مدرسين الطالب، أو الطالب نفسه من غير ما يغير حالته أو إيميله أو مدرسينه
      allow update: if isAdmin()
        || (isTeacher() && request.auth.uid in resource.data.teacherIds)
        || (request.auth != null && request.auth.uid == studentId
            && request.resource.data.status == resource.data.status
            && request.resource.data.email == resource.data.email
            && request.resource.data.teacherIds == resource.data.teacherIds);
      // حذف: الأدمن، أو مدرس من مدرسين الطالب، أو الطالب نفسه (بشيل ملفه القديم عند أول دخول).
      // المطابقة بالإيميل برضه قبل الربط بس
      allow delete: if isAdmin()
        || (isTeacher() && request.auth.uid in resource.data.teacherIds)
        || (request.auth != null && request.auth.token.email == resource.data.email
            && resource.data.authLinked != true);
    }
  }
}
